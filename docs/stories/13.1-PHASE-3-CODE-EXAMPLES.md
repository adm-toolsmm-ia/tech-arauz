## 1. ProcessSlaModal — Complete Implementation Example

### Component Structure (Use as Template)

```typescript
'use client';

import React, { useState, useEffect } from 'react';
import { Sheet, SheetContent, SheetHeader, SheetTitle } from '@/components/ui/sheet';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Textarea } from '@/components/ui/textarea';
import { Label } from '@/components/ui/label';
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select';
import { Card, CardContent } from '@/components/ui/card';
import { AlertTriangle, Save, Loader2, X } from 'lucide-react';
import { toast } from 'sonner';
import type { OrgProcessSla } from '@/types/organization';
import {
  createProcessSlaAction,
  updateProcessSlaAction,
} from '@/app/actions/organization';

interface ProcessSlaModalProps {
  isOpen: boolean;
  processId: string;
  sla?: OrgProcessSla; // undefined for create, populated for edit
  onClose: () => void;
  onSave?: () => void;
}

type FormData = {
  sla_type: 'execution_time' | 'quality' | 'availability';
  target_value: number;
  unit: string;
  period: 'daily' | 'weekly' | 'monthly';
  description?: string | null;
};

const defaultFormData: FormData = {
  sla_type: 'execution_time',
  target_value: 0,
  unit: 'minutes',
  period: 'daily',
  description: null,
};

export function ProcessSlaModal({
  isOpen,
  processId,
  sla,
  onClose,
  onSave,
}: ProcessSlaModalProps) {
  const [formData, setFormData] = useState<FormData>(defaultFormData);
  const [isDirty, setIsDirty] = useState(false);
  const [isSaving, setIsSaving] = useState(false);
  const [errors, setErrors] = useState<Record<string, string>>({});

  const mode = sla ? 'edit' : 'create';
  const title = mode === 'create' ? 'Nova Definição de SLA' : 'Editar Definição de SLA';

  // Initialize form from SLA data
  useEffect(() => {
    if (sla && isOpen) {
      setFormData({
        sla_type: sla.sla_type as any,
        target_value: sla.target_value,
        unit: sla.unit,
        period: sla.period as any,
        description: sla.description || null,
      });
      setIsDirty(false);
      setErrors({});
    } else if (mode === 'create' && isOpen) {
      setFormData(defaultFormData);
      setIsDirty(false);
      setErrors({});
    }
  }, [sla, isOpen, mode]);

  const handleChange = (field: keyof FormData, value: any) => {
    setFormData((prev) => ({ ...prev, [field]: value }));
    setIsDirty(true);
  };

  const handleSave = async () => {
    // Validation
    const newErrors: Record<string, string> = {};

    if (!formData.target_value || formData.target_value <= 0) {
      newErrors.target_value = 'Valor alvo deve ser maior que 0';
    }

    if (Object.keys(newErrors).length > 0) {
      setErrors(newErrors);
      toast.error('❌ Corrija os erros antes de salvar');
      return;
    }

    setIsSaving(true);
    try {
      const result =
        mode === 'create'
          ? await createProcessSlaAction(processId, formData)
          : await updateProcessSlaAction(sla!.id, formData);

      if (result.success) {
        toast.success(`✅ SLA ${mode === 'create' ? 'criada' : 'atualizada'} com sucesso!`);
        setIsDirty(false);
        onSave?.();
        onClose();
      } else {
        setErrors(result.field_errors || {});
        toast.error(result.message || 'Erro ao salvar SLA');
      }
    } catch (error) {
      toast.error(`❌ Erro: ${error instanceof Error ? error.message : 'desconhecido'}`);
    } finally {
      setIsSaving(false);
    }
  };

  return (
    <Sheet open={isOpen} onOpenChange={(open) => !open && onClose()}>
      <SheetContent side="right" className="w-full max-w-2xl p-0 overflow-hidden flex flex-col">
        {/* Header */}
        <SheetHeader className="border-b px-6 py-4">
          <div className="flex items-center justify-between">
            <SheetTitle>{title}</SheetTitle>
            <Button
              variant="ghost"
              size="sm"
              onClick={onClose}
              className="h-auto p-0"
              aria-label="Fechar modal"
            >
              <X className="h-4 w-4" />
            </Button>
          </div>
        </SheetHeader>

        {/* Dirty State Alert */}
        {isDirty && (
          <Card className="m-4 border-yellow-200 bg-yellow-50 dark:border-yellow-800 dark:bg-yellow-950">
            <CardContent className="pt-4">
              <div className="flex items-center gap-2">
                <AlertTriangle className="h-4 w-4 text-yellow-600 dark:text-yellow-400" />
                <p className="text-sm text-yellow-800 dark:text-yellow-200">
                  Você tem alterações não salvas.
                </p>
              </div>
            </CardContent>
          </Card>
        )}

        {/* Form Content */}
        <div className="flex-1 overflow-y-auto px-6 py-4 space-y-6">
          {/* SLA Type */}
          <div>
            <Label htmlFor="sla_type">Tipo de SLA *</Label>
            <Select
              value={formData.sla_type}
              onValueChange={(value) => handleChange('sla_type', value as any)}
              disabled={isSaving}
            >
              <SelectTrigger id="sla_type" aria-invalid={!!errors.sla_type}>
                <SelectValue />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="execution_time">Tempo de Execução</SelectItem>
                <SelectItem value="quality">Qualidade</SelectItem>
                <SelectItem value="availability">Disponibilidade</SelectItem>
              </SelectContent>
            </Select>
            {errors.sla_type && (
              <p id="sla_type-error" className="text-xs text-red-500 mt-1" role="alert">
                {errors.sla_type}
              </p>
            )}
          </div>

          {/* Target Value */}
          <div>
            <Label htmlFor="target_value">Valor Alvo *</Label>
            <Input
              id="target_value"
              type="number"
              value={formData.target_value || ''}
              onChange={(e) => handleChange('target_value', Number(e.target.value))}
              placeholder="ex.: 30"
              disabled={isSaving}
              aria-invalid={!!errors.target_value}
              aria-describedby={errors.target_value ? 'target_value-error' : undefined}
              className={errors.target_value ? 'border-red-500' : ''}
            />
            {errors.target_value && (
              <p id="target_value-error" className="text-xs text-red-500 mt-1" role="alert">
                {errors.target_value}
              </p>
            )}
          </div>

          {/* Grid: Unit + Period */}
          <div className="grid grid-cols-2 gap-4">
            {/* Unit */}
            <div>
              <Label htmlFor="unit">Unidade *</Label>
              <Select
                value={formData.unit}
                onValueChange={(value) => handleChange('unit', value)}
                disabled={isSaving}
              >
                <SelectTrigger id="unit">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="minutes">Minutos</SelectItem>
                  <SelectItem value="percent">Percentual (%)</SelectItem>
                  <SelectItem value="occurrences">Ocorrências</SelectItem>
                </SelectContent>
              </Select>
            </div>

            {/* Period */}
            <div>
              <Label htmlFor="period">Período *</Label>
              <Select
                value={formData.period}
                onValueChange={(value) => handleChange('period', value as any)}
                disabled={isSaving}
              >
                <SelectTrigger id="period">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="daily">Diário</SelectItem>
                  <SelectItem value="weekly">Semanal</SelectItem>
                  <SelectItem value="monthly">Mensal</SelectItem>
                </SelectContent>
              </Select>
            </div>
          </div>

          {/* Description */}
          <div>
            <Label htmlFor="description">Descrição (opcional)</Label>
            <Textarea
              id="description"
              value={formData.description || ''}
              onChange={(e) => handleChange('description', e.target.value || null)}
              placeholder="Descreva o objetivo desta SLA..."
              rows={3}
              disabled={isSaving}
            />
          </div>
        </div>

        {/* Footer: Action Buttons */}
        <div className="border-t px-6 py-4 flex justify-end gap-2">
          <Button
            variant="outline"
            onClick={onClose}
            disabled={isSaving}
          >
            Cancelar
          </Button>
          <Button
            onClick={handleSave}
            disabled={isSaving}
            className="gap-2"
          >
            {isSaving ? (
              <>
                <Loader2 className="w-4 h-4 animate-spin" />
                Salvando...
              </>
            ) : (
              <>
                <Save className="w-4 h-4" />
                Salvar
              </>
            )}
          </Button>
        </div>
      </SheetContent>
    </Sheet>
  );
}
```

## 2. ProcessSlaList — Complete Implementation Example

```typescript
'use client';

import React from 'react';
import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from '@/components/ui/table';
import { Button } from '@/components/ui/button';
import { Edit2, Trash2 } from 'lucide-react';
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogHeader,
  AlertDialogTitle,
} from '@/components/ui/alert-dialog';
import { toast } from 'sonner';
import type { OrgProcessSla } from '@/types/organization';
import { deleteProcessSlaAction } from '@/app/actions/organization';
import { EmptyState } from '@/components/ui/EmptyState';

interface ProcessSlaListProps {
  processId: string;
  slas: OrgProcessSla[];
  onEdit: (sla: OrgProcessSla) => void;
  onDelete: (slaId: string) => void;
  onRefresh: () => void;
}

const slaTypeLabels: Record<string, string> = {
  execution_time: 'Tempo de Execução',
  quality: 'Qualidade',
  availability: 'Disponibilidade',
};

const unitLabels: Record<string, string> = {
  minutes: 'Minutos',
  percent: '%',
  occurrences: 'Ocorrências',
};

const periodLabels: Record<string, string> = {
  daily: 'Diário',
  weekly: 'Semanal',
  monthly: 'Mensal',
};

export function ProcessSlaList({
  processId,
  slas,
  onEdit,
  onDelete,
  onRefresh,
}: ProcessSlaListProps) {
  const [deletingId, setDeletingId] = React.useState<string | null>(null);
  const [isDeleting, setIsDeleting] = React.useState(false);

  const handleConfirmDelete = async (slaId: string) => {
    setIsDeleting(true);
    try {
      const result = await deleteProcessSlaAction(slaId);
      if (result.success) {
        toast.success('✅ SLA removida com sucesso!');
        onDelete(slaId);
        setDeletingId(null);
        onRefresh();
      } else {
        toast.error(result.message || 'Erro ao remover SLA');
      }
    } catch (error) {
      toast.error(`❌ Erro: ${error instanceof Error ? error.message : 'desconhecido'}`);
    } finally {
      setIsDeleting(false);
    }
  };

  if (slas.length === 0) {
    return (
      <EmptyState
        icon="📊"
        title="Nenhuma SLA definida"
        description="Clique em '+ Nova SLA' para criar a primeira definição de SLA para este processo."
      />
    );
  }

  return (
    <>
      <div className="border rounded-lg overflow-hidden">
        <Table>
          <TableHeader>
            <TableRow className="bg-muted/50">
              <TableHead>Tipo</TableHead>
              <TableHead className="text-right">Valor Alvo</TableHead>
              <TableHead>Período</TableHead>
              <TableHead>Descrição</TableHead>
              <TableHead className="text-right">Ações</TableHead>
            </TableRow>
          </TableHeader>
          <TableBody>
            {slas.map((sla) => (
              <TableRow key={sla.id} className="hover:bg-muted/50">
                <TableCell className="font-medium">
                  {slaTypeLabels[sla.sla_type] || sla.sla_type}
                </TableCell>
                <TableCell className="text-right">
                  <span className="font-mono">
                    {sla.target_value} {unitLabels[sla.unit] || sla.unit}
                  </span>
                </TableCell>
                <TableCell>
                  {periodLabels[sla.period] || sla.period}
                </TableCell>
                <TableCell className="text-sm text-muted-foreground max-w-xs truncate">
                  {sla.description || '—'}
                </TableCell>
                <TableCell className="text-right">
                  <div className="flex justify-end gap-2">
                    <Button
                      variant="ghost"
                      size="sm"
                      onClick={() => onEdit(sla)}
                      className="gap-1"
                      aria-label={`Editar SLA ${slaTypeLabels[sla.sla_type]}`}
                    >
                      <Edit2 className="w-4 h-4" />
                      <span className="hidden sm:inline">Editar</span>
                    </Button>
                    <Button
                      variant="ghost"
                      size="sm"
                      className="text-destructive hover:text-destructive gap-1"
                      onClick={() => setDeletingId(sla.id)}
                      aria-label={`Remover SLA ${slaTypeLabels[sla.sla_type]}`}
                    >
                      <Trash2 className="w-4 h-4" />
                      <span className="hidden sm:inline">Remover</span>
                    </Button>
                  </div>
                </TableCell>
              </TableRow>
            ))}
          </TableBody>
        </Table>
      </div>

      {/* Delete Confirmation Dialog */}
      <AlertDialog open={!!deletingId} onOpenChange={(open) => !open && setDeletingId(null)}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Remover SLA?</AlertDialogTitle>
            <AlertDialogDescription>
              Tem certeza que deseja remover esta definição de SLA? Esta ação não pode ser desfeita.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <div className="flex justify-end gap-2">
            <AlertDialogCancel disabled={isDeleting}>Cancelar</AlertDialogCancel>
            <AlertDialogAction
              onClick={() => deletingId && handleConfirmDelete(deletingId)}
              disabled={isDeleting}
              className="bg-destructive hover:bg-destructive/90"
            >
              {isDeleting ? 'Removendo...' : 'Remover'}
            </AlertDialogAction>
          </div>
        </AlertDialogContent>
      </AlertDialog>
    </>
  );
}
```

```typescript
// In ProcessCockpit360.tsx

import { ProcessSlaModal } from './ProcessSlaModal';
import { ProcessSlaList } from './ProcessSlaList';

export function ProcessCockpit360({
  process,
  // ... other props
}: ProcessCockpit360Props) {
  // ... existing state
  const [slaModalOpen, setSlaModalOpen] = useState(false);
  const [editingSla, setEditingSla] = useState<OrgProcessSla | null>(null);
  const [slas, setSlas] = useState<OrgProcessSla[]>([]);
  const [loadingSlas, setLoadingSlas] = useState(false);

  // Fetch SLAs on component load
  useEffect(() => {
    if (!process?.id) return;
    const loadSlas = async () => {
      setLoadingSlas(true);
      try {
        const result = await getProcessSLAsAction(process.id);

          setSlas(result.data);
        }
      } catch (error) {
        console.error('Erro ao carregar SLAs:', error);
      } finally {
        setLoadingSlas(false);
      }
    };
    loadSlas();
  }, [process?.id]);

  return (
    <div className="space-y-6">
      <Tabs defaultValue="principal" className="w-full">
        {/* ... existing tabs ... */}

        {/* NEW: SLA Management Tab */}
        <TabsTrigger
          value="slas"
          className="rounded-none border-b-2 border-transparent px-4 py-2.5 data-[state=active]:border-primary data-[state=active]:bg-transparent"
        >
          <BarChart3 className="mr-2 size-4" />
          SLAs
          {slas.length > 0 && (
            <span className="ml-2 text-xs text-muted-foreground">({slas.length})</span>
          )}
        </TabsTrigger>

        {/* ... existing tab contents ... */}

        {/* NEW: SLA Management Tab Content */}
        <TabsContent value="slas" className="mt-6 space-y-4">
          <div className="flex justify-end">
            <Button
              onClick={() => {
                setEditingSla(null);
                setSlaModalOpen(true);
              }}
              className="gap-2"
            >
              <Plus className="size-4" />
              Nova SLA
            </Button>
          </div>

          <ProcessSlaList
            processId={process.id}
            slas={slas}
            onEdit={(sla) => {
              setEditingSla(sla);
              setSlaModalOpen(true);
            }}
            onDelete={(slaId) => {
              setSlas((prev) => prev.filter((s) => s.id !== slaId));
            }}
            onRefresh={async () => {
              const result = await getProcessSLAsAction(process.id);
              if (result.success && result.data) {
                setSlas(result.data);
              }
            }}
          />
        </TabsContent>
      </Tabs>

      {/* SLA Modal */}
      <ProcessSlaModal
        isOpen={slaModalOpen}
        processId={process.id}
        sla={editingSla || undefined}
        onClose={() => {
          setSlaModalOpen(false);
          setEditingSla(null);
        }}
        onSave={async () => {
          const result = await getProcessSLAsAction(process.id);
          if (result.success && result.data) {
            setSlas(result.data);
          }
        }}
      />
    </div>
  );
}
```

## 4. Accessibility Best Practices Used

### Label Association
```typescript
// ✅ GOOD: htmlFor connects to input id
<Label htmlFor="sla_type">Tipo de SLA *</Label>
<Select>
  <SelectTrigger id="sla_type">
    <SelectValue />
  </SelectTrigger>
</Select>

// ❌ BAD: No htmlFor/id connection
<Label>Tipo de SLA *</Label>
<Select>
  <SelectTrigger>
    <SelectValue />
  </SelectTrigger>
</Select>
```

### ARIA Attributes for Errors
```typescript
// ✅ GOOD: aria-invalid + aria-describedby
<Input
  id="target_value"
  aria-invalid={!!errors.target_value}
  aria-describedby={errors.target_value ? 'target_value-error' : undefined}
  className={errors.target_value ? 'border-red-500' : ''}
/>
{errors.target_value && (
  <p id="target_value-error" className="text-xs text-red-500 mt-1" role="alert">
    {errors.target_value}
  </p>
)}

// ❌ BAD: Just visual error, not semantic
<Input className={errors.target_value ? 'border-red-500' : ''} />
<p>{errors.target_value}</p>
```

## 8. Common Pitfalls to Avoid

### ❌ DON'T: Missing Labels
```typescript
// ❌ BAD
<Input placeholder="Nome" />

// ✅ GOOD
<Label htmlFor="name">Nome *</Label>
<Input id="name" placeholder="Digite o nome" />
```

### ❌ DON'T: Inaccessible Icons
```typescript
// ❌ BAD: Just an icon, no label
<Button>
  <Trash2 className="w-4 h-4" />
</Button>

// ✅ GOOD: Icon + aria-label or text
<Button aria-label="Remover SLA">
  <Trash2 className="w-4 h-4" />
</Button>
// or
<Button>
  <Trash2 className="w-4 h-4" />
  <span className="hidden sm:inline">Remover</span>
</Button>
```

### ❌ DON'T: Disabled Without Reason
```typescript
// ❌ BAD: Disabled but no explanation
<Button disabled>Salvar</Button>

// ✅ GOOD: Loading indicator
<Button disabled={isSaving}>
  {isSaving ? (
    <>
      <Loader2 className="animate-spin" />
      Salvando...
    </>
  ) : (
    'Salvar'
  )}
</Button>
```

### ❌ DON'T: Trap Focus in Modal
```typescript
// ✅ GOOD: Sheet component handles focus trap
<Sheet open={isOpen}>
  <SheetContent>
    {/* Focus is automatically trapped and restored */}
  </SheetContent>
</Sheet>

// Not needed - shadcn Sheet handles it!
```

## 9. Server Action Examples

```typescript
// In src/app/actions/organization.ts

/**
 * Create SLA for process
 * Story 13.1: Process SLA Management UI
 */
export async function createProcessSlaAction(
  processId: string,
  payload: {
    sla_type: 'execution_time' | 'quality' | 'availability';
    target_value: number;
    unit: string;
    period: 'daily' | 'weekly' | 'monthly';
    description?: string;
  }
): Promise<OrgActionResult> {
  const ctx = await getAuthContext();
  if ('error' in ctx) {
    return { success: false, message: ctx.error };
  }

  // Validation
  if (!processId || !payload.sla_type || payload.target_value === undefined) {
    return {
      success: false,
      message: 'Missing required fields',
      field_errors: {
        sla_type: !payload.sla_type ? 'Required' : '',
        target_value: payload.target_value === undefined ? 'Required' : '',
      },
    };
  }

  if (payload.target_value <= 0) {
    return {
      success: false,
      message: 'Target value must be greater than 0',
      field_errors: { target_value: 'Must be greater than 0' },
    };
  }

  const { error } = await ctx.supabase.from('org_process_slas').insert({
    process_id: processId,
    tenant_id: ctx.tenantId,
    sla_type: payload.sla_type,
    target_value: payload.target_value,
    unit: payload.unit,
    period: payload.period,
    description: payload.description || null,
  });

  if (error) {
    return {
      success: false,
      message: `Error creating SLA: ${error.message}`,
    };
  }

  revalidatePath('/organizacao');
  return {
    success: true,
    message: 'SLA created successfully!',
  };
}
```
