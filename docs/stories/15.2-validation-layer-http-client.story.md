## 🏗️ Arquitetura

### Error Handling Pattern

```typescript
// ANTES (sem validação):
const data = await response.json();
if (!data || typeof data !== 'object') {
  throw createError('INVALID_RESPONSE', 'Invalid JSON response', requestId);
}
// Assume data é seguro — NÃO É!

// DEPOIS (com validação):
try {
  const data = await response.json();
  const validatedResponse = ExportarDadosResponseSchema.parse(data);
  return validatedResponse;
} catch (error) {
  if (error instanceof z.ZodError) {
    const details = error.errors.map(e => ({
      path: e.path.join('.'),
      message: e.message,
      code: e.code,
    }));
    const msg = `API contract violation: ${JSON.stringify(details)}`;
    log('ERROR', `Validation failed: ${msg}`, { requestId, validationErrors: details });
    throw createError('INVALID_RESPONSE', msg, requestId);
  }
  // Outros errors (JSON parse, etc.)
  throw classifyError(error, requestId);
}
```

### Integration Points

**In `executeWithRetry()`:**
```typescript
const response = await fetch(url, fetchOptions);
const data = await response.json();

// Validate based on method
const validatedData = method === 'POST'
  ? ExportarDadosResponseSchema.parse(data)
  : ExportarDadosResponseSchema.parse(data); // Same for GET

// Check Espaider error status AFTER validation
if (validatedData.Situacao === 'E') {
  throw createError(
    'INVALID_RESPONSE',
    `Espaider error: ${validatedData.MensagemRetorno || 'Unknown'}`,
    requestId,
  );
}

recordSuccess();
return validatedData;
```
