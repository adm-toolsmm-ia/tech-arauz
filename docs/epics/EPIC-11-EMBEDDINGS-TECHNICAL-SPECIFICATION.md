### Migration File: `071_add_embeddings_to_knowledge_entries.sql`

```sql
-- Enable pgvector extension (idempotent)
CREATE EXTENSION IF NOT EXISTS vector;

-- Add embedding column to org_knowledge_entries
-- Dimension: 1536 (standard for OpenAI text-embedding-3-small)
ALTER TABLE public.org_knowledge_entries
ADD COLUMN IF NOT EXISTS embedding vector(1536) DEFAULT NULL;

-- Create IVFFlat index for similarity search
-- lists=100: optimal for 100k+ rows (formula: sqrt(rows) for large datasets)
-- vector_cosine_ops: cosine distance metric for semantic similarity
CREATE INDEX IF NOT EXISTS idx_org_knowledge_entries_embedding
ON public.org_knowledge_entries USING ivfflat (embedding vector_cosine_ops)
WITH (lists = 100);

-- Documentation
COMMENT ON COLUMN public.org_knowledge_entries.embedding
  IS 'Vector embedding (1536-dim) generated by OpenAI text-embedding-3-small.
  Used for semantic similarity search via cosine distance.
  Nullable to support incremental embedding generation.';

COMMENT ON INDEX idx_org_knowledge_entries_embedding
  IS 'IVFFlat index for semantic similarity search.
  Use with probes=10 in queries for best speed/recall balance.
  Rebuild if dataset grows beyond 1M rows.';

-- Create function for semantic similarity search
-- Returns top-k most similar entries by cosine distance
CREATE OR REPLACE FUNCTION search_knowledge_semantic(
  query_embedding vector(1536),
  tenant_id_param uuid,
  k integer DEFAULT 10
)
RETURNS TABLE (
  id uuid,
  title text,
  content text,
  type text,
  source_url text,
  similarity_score float8
) AS $$
  SELECT
    ke.id,
    ke.title,
    ke.content,
    ke.type,
    ke.source_url,
    (1 - (ke.embedding <=> query_embedding)) as similarity_score
  FROM public.org_knowledge_entries ke
  WHERE ke.tenant_id = tenant_id_param
    AND ke.embedding IS NOT NULL
  ORDER BY ke.embedding <=> query_embedding
  LIMIT k;
$$ LANGUAGE sql STABLE;

COMMENT ON FUNCTION search_knowledge_semantic(vector, uuid, integer)
  IS 'Semantic similarity search on org_knowledge_entries.
  Returns top-k entries most similar to the query embedding.
  Cosine distance metric: (1 - <=> operator) gives similarity [0, 1].';

-- Optional: Create batch status tracking table
CREATE TABLE IF NOT EXISTS public.embedding_batch_log (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES public.tenants(id) ON DELETE CASCADE,
  batch_type TEXT NOT NULL CHECK (batch_type IN ('initial', 'incremental', 'rebuild')),
  started_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  completed_at TIMESTAMPTZ,
  total_entries INTEGER,
  processed_entries INTEGER,
  failed_entries INTEGER,
  api_cost NUMERIC(10, 4),
  error_message TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_embedding_batch_log_tenant ON public.embedding_batch_log(tenant_id);
CREATE INDEX idx_embedding_batch_log_completed ON public.embedding_batch_log(completed_at);

COMMENT ON TABLE public.embedding_batch_log
  IS 'Tracks batch embedding generation jobs for monitoring and cost analysis.';
```

## 🔌 Server Actions

### File: `src/app/actions/knowledge.ts` (additions)

```typescript
'use server';

import { createClient } from '@/lib/supabase/server';
import { OrgKnowledgeEntry, OrgKnowledgeSearchResult, OrgActionResult } from '@/types';
import { getOrganizationContext } from '@/lib/auth';

/**
 * Semantic search across organization knowledge base
 *
 * 1. Convert query text to embedding via OpenAI API
 * 2. Call database RPC function with embedding
 * 3. Return top-k results with similarity scores
 */
export async function semanticSearchKnowledgeAction(
  query: string,
  options?: {
    limit?: number;
    minSimilarity?: number;
  }
): Promise<OrgActionResult<OrgKnowledgeSearchResult[]>> {
  try {
    const { tenantId, userId } = await getOrganizationContext();
    const supabase = createClient();

    const limit = options?.limit ?? 10;
    const minSimilarity = options?.minSimilarity ?? 0.5;

    // Step 1: Generate embedding for query
    const queryEmbedding = await generateEmbedding(query);
    if (!queryEmbedding) {
      return {
        success: false,
        error: 'Failed to generate query embedding',
      };
    }

    // Step 2: Call semantic search RPC function
    const { data, error } = await supabase.rpc(
      'search_knowledge_semantic',
      {
        query_embedding: queryEmbedding,
        tenant_id_param: tenantId,
        k: limit,
      },
      { head: false }
    );

    if (error) {
      console.error('[semanticSearchKnowledgeAction] RPC error:', error);
      return {
        success: false,
        error: `Semantic search failed: ${error.message}`,
      };
    }

    // Step 3: Filter by minimum similarity threshold
    const results = (data || []).filter(
      (result: any) => result.similarity_score >= minSimilarity
    );

    // Log search metric
    await logSearchMetric(tenantId, query, results.length);

    return {
      success: true,
      data: results as OrgKnowledgeSearchResult[],
    };
  } catch (error) {
    console.error('[semanticSearchKnowledgeAction] Error:', error);
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error',
    };
  }
}

/**
 * Generate embedding for text using OpenAI API
 * Returns 1536-dimensional vector or null on failure
 */
async function generateEmbedding(text: string): Promise<number[] | null> {
  try {
    if (!process.env.OPENAI_API_KEY) {
      throw new Error('OPENAI_API_KEY not configured');
    }

    // Truncate to reasonable length (OpenAI has token limits)
    const truncatedText = text.substring(0, 8000);

    const response = await fetch('https://api.openai.com/v1/embeddings', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${process.env.OPENAI_API_KEY}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        model: 'text-embedding-3-small',
        input: truncatedText,
        dimensions: 1536, // Standard dimension
      }),
    });

    if (!response.ok) {
      throw new Error(`OpenAI API error: ${response.statusText}`);
    }

    const result = await response.json();
    return result.data?.[0]?.embedding ?? null;
  } catch (error) {
    console.error('[generateEmbedding] Error:', error);
    return null;
  }
}

/**
 * Log search metrics for monitoring and analytics
 */
async function logSearchMetric(
  tenantId: string,
  query: string,
  resultCount: number
): Promise<void> {
  try {
    // Optional: Store in analytics table for future analysis
    // For now, just log to console/monitoring system
    console.log('[semanticSearchKnowledgeAction]', {
      tenantId,
      queryLength: query.length,
      resultCount,
      timestamp: new Date().toISOString(),
    });
  } catch (error) {
    // Non-critical: don't fail if logging fails
    console.warn('[logSearchMetric] Failed to log metric:', error);
  }
}

/**
 * Batch generate embeddings for all knowledge entries
 * Used for initial population and periodic refreshes
 *
 * Returns: { processedCount, failedCount, apiCost }
 */
export async function generateKnowledgeEmbeddingsBatchAction(
  tenantId?: string
): Promise<OrgActionResult<{ processedCount: number; failedCount: number; apiCost: number }>> {
  try {
    const { tenantId: contextTenantId } = await getOrganizationContext();
    const targetTenantId = tenantId || contextTenantId;

    // Only allow admin to trigger batch jobs
    const isAdmin = await checkAdminAccess(targetTenantId);
    if (!isAdmin) {
      return {
        success: false,
        error: 'Only admins can trigger batch embedding generation',
      };
    }

    const supabase = createClient();

    // Fetch all knowledge entries without embeddings
    const { data: entries, error: fetchError } = await supabase
      .from('org_knowledge_entries')
      .select('id, title, content')
      .eq('tenant_id', targetTenantId)
      .is('embedding', null)
      .limit(1000);

    if (fetchError) throw fetchError;
    if (!entries || entries.length === 0) {
      return {
        success: true,
        data: { processedCount: 0, failedCount: 0, apiCost: 0 },
      };
    }

    let processedCount = 0;
    let failedCount = 0;
    let apiCost = 0;

    // Generate embeddings in batches
    const BATCH_SIZE = 100;
    for (let i = 0; i < entries.length; i += BATCH_SIZE) {
      const batch = entries.slice(i, i + BATCH_SIZE);

      for (const entry of batch) {
        try {
          const embedding = await generateEmbedding(entry.content);
          if (!embedding) {
            failedCount++;
            continue;
          }

          // Update entry with embedding
          const { error: updateError } = await supabase
            .from('org_knowledge_entries')
            .update({
              embedding,
              embedding_generated_at: new Date().toISOString(),
              embedding_model: 'text-embedding-3-small',
            })
            .eq('id', entry.id);

          if (updateError) {
            console.error(`Failed to update entry ${entry.id}:`, updateError);
            failedCount++;
          } else {
            processedCount++;
            // Estimate cost: 1536 dimensions = ~200 tokens per entry (worst case)
            apiCost += 0.00002; // text-embedding-3-small cost per token
          }
        } catch (error) {
          console.error(`Error embedding ${entry.id}:`, error);
          failedCount++;
        }
      }

      // Log progress
      console.log(`[generateKnowledgeEmbeddingsBatchAction] Progress: ${processedCount + failedCount}/${entries.length}`);
    }

    // Log batch completion
    await logBatchCompletion(targetTenantId, processedCount, failedCount, apiCost);

    return {
      success: true,
      data: { processedCount, failedCount, apiCost },
    };
  } catch (error) {
    console.error('[generateKnowledgeEmbeddingsBatchAction] Error:', error);
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error',
    };
  }
}

async function checkAdminAccess(tenantId: string): Promise<boolean> {
  // Implementation: Check if current user is admin for tenant
  const { userId } = await getOrganizationContext();
  const supabase = createClient();

  const { data } = await supabase
    .from('profiles')
    .select('role')
    .eq('tenant_id', tenantId)
    .eq('id', userId)
    .single();

  return data?.role === 'admin';
}

async function logBatchCompletion(
  tenantId: string,
  processedCount: number,
  failedCount: number,
  apiCost: number
): Promise<void> {
  try {
    const supabase = createClient();
    await supabase.from('embedding_batch_log').insert({
      tenant_id: tenantId,
      batch_type: 'incremental',
      total_entries: processedCount + failedCount,
      processed_entries: processedCount,
      failed_entries: failedCount,
      api_cost: apiCost,
      completed_at: new Date().toISOString(),
    });
  } catch (error) {
    console.warn('[logBatchCompletion] Failed to log batch:', error);
  }
}
```

```typescript
import { describe, it, expect, beforeEach, vi } from 'vitest';
import {
  semanticSearchKnowledgeAction,
  generateKnowledgeEmbeddingsBatchAction,
} from '@/app/actions/knowledge';

describe('Knowledge Semantic Search', () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe('semanticSearchKnowledgeAction', () => {
    it('should return relevant results for semantic query', async () => {
      const result = await semanticSearchKnowledgeAction('responsible roles in activities');
      expect(result.success).toBe(true);
      expect(Array.isArray(result.data)).toBe(true);
      expect(result.data?.length).toBeGreaterThan(0);
    });

    it('should respect similarity threshold', async () => {
      const result = await semanticSearchKnowledgeAction('gibberish xyz 123', {
        minSimilarity: 0.9,
      });
      expect(result.success).toBe(true);
      // Should return empty or very few results for nonsense query
      expect(result.data?.length ?? 0).toBeLessThan(3);
    });

    it('should handle API errors gracefully', async () => {
      vi.mock('@/lib/supabase/server', () => ({
        createClient: () => ({
          rpc: () => Promise.reject(new Error('API Error')),
        }),
      }));

      const result = await semanticSearchKnowledgeAction('test');
      expect(result.success).toBe(false);
      expect(result.error).toBeDefined();
    });

    it('should validate similarity scores', async () => {
      const result = await semanticSearchKnowledgeAction('test query');
      if (result.success && result.data) {
        result.data.forEach(item => {
          expect(item.similarity_score).toBeGreaterThanOrEqual(0);
          expect(item.similarity_score).toBeLessThanOrEqual(1);
        });
      }
    });
  });

  describe('generateKnowledgeEmbeddingsBatchAction', () => {
    it('should process embeddings for knowledge entries', async () => {
      const result = await generateKnowledgeEmbeddingsBatchAction();
      expect(result.success).toBe(true);
      expect(result.data?.processedCount).toBeGreaterThanOrEqual(0);
      expect(result.data?.failedCount).toBeGreaterThanOrEqual(0);
    });

    it('should estimate API costs correctly', async () => {
      const result = await generateKnowledgeEmbeddingsBatchAction();
      if (result.success && result.data) {
        expect(result.data.apiCost).toBeGreaterThanOrEqual(0);
        expect(result.data.apiCost).toBeLessThan(100); // Sanity check
      }
    });

    it('should skip already-embedded entries', async () => {
      const result1 = await generateKnowledgeEmbeddingsBatchAction();
      const count1 = result1.data?.processedCount ?? 0;

      const result2 = await generateKnowledgeEmbeddingsBatchAction();
      const count2 = result2.data?.processedCount ?? 0;

      expect(count2).toBeLessThanOrEqual(count1);
    });
  });
});
```
