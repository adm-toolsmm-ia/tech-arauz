## Notas Técnicas

### Padrão de Tab "Rotinas" em ProcessCockpit360

```typescript
const [routines, setRoutines] = useState<OrgRoutine[]>([]);
const [loadingRoutines, setLoadingRoutines] = useState(false);

useEffect(() => {
  if (!process?.id) return;
  setLoadingRoutines(true);
  (async () => {
    const result = await getRoutinesByProcess(process.id);
    setRoutines(result || []);
    setLoadingRoutines(false);
  })();
}, [process?.id]);

// Na renderização:
<TabsContent value="rotinas">
  <div className="space-y-4">
    {loadingRoutines ? (
      <Skeleton className="h-20" />
    ) : (
      <div className="grid gap-2">
        {routines.map(routine => (
          <OrgEntityCard
            key={routine.id}
            title={routine.name}
            subtitle={routine.description}
            badge={`${routine.activities_count} atividades`}
            onClick={() => onSelectRoutine?.(routine)}
          />
        ))}
      </div>
    )}
  </div>
</TabsContent>
```

### State Management com 3 níveis

```typescript
// Em areas-content.tsx
const [selectedArea, setSelectedArea] = useState<OrgArea | null>(null);
const [selectedNucleus, setSelectedNucleus] = useState<OrgNucleus | null>(null);
const [selectedProcess, setSelectedProcess] = useState<OrgProcess | null>(null);
const [selectedRoutine, setSelectedRoutine] = useState<OrgRoutine | null>(null);

// Invariante: fechar pai reseta filhos
const handleSelectArea = (area: OrgArea | null) => {
  setSelectedArea(area);
  if (!area) {
    setSelectedNucleus(null);
    setSelectedProcess(null);
    setSelectedRoutine(null);
  }
};

const handleSelectNucleus = (nucleus: OrgNucleus | null) => {
  setSelectedNucleus(nucleus);
  if (!nucleus) {
    setSelectedProcess(null);
    setSelectedRoutine(null);
  }
};

const handleSelectProcess = (process: OrgProcess | null) => {
  setSelectedProcess(process);
  if (!process) {
    setSelectedRoutine(null);
  }
};
```

<ContextPanel

      { label: "Processo", onClick: () => handleSelectProcess(selectedProcess) },
