### Gap 3: SLA Management (New)

**New Server Actions (organization.ts):**

```typescript
/**
 * Create SLA for process
 * Story 13.1: Process SLA Management UI
 */
export async function createProcessSlaAction(
  processId: string,
  payload: {
    sla_type: 'execution_time' | 'quality' | 'availability';
    target_value: number;
    unit: string; // 'minutes', 'percent', 'occurrences'
    period: 'daily' | 'weekly' | 'monthly';
    description?: string;
  }
): Promise<OrgActionResult> {
  const ctx = await getAuthContext();
  if ('error' in ctx) return { success: false, message: ctx.error };

  // Validation
  if (!processId || !payload.sla_type || payload.target_value === undefined) {
    return { success: false, message: 'Missing required fields' };
  }

  const { error } = await ctx.supabase.from('org_process_slas').insert({
    process_id: processId,
    tenant_id: ctx.tenantId,
    sla_type: payload.sla_type,
    target_value: payload.target_value,
    unit: payload.unit,
    period: payload.period,
    description: payload.description || null,
  });

  if (error) {
    return { success: false, message: `Error creating SLA: ${error.message}` };
  }

  revalidatePath('/organizacao');
  return { success: true, message: 'SLA created successfully!' };
}

/**
 * Update SLA
 * Story 13.1: Process SLA Management UI
 */
export async function updateProcessSlaAction(
  slaId: string,
  updates: Partial<{
    target_value: number;
    unit: string;
    period: string;
    description: string;
  }>
): Promise<OrgActionResult> {
  const ctx = await getAuthContext();
  if ('error' in ctx) return { success: false, message: ctx.error };

  const { error } = await ctx.supabase
    .from('org_process_slas')
    .update(updates)
    .eq('id', slaId)
    .eq('tenant_id', ctx.tenantId);

  if (error) {
    return { success: false, message: `Error updating SLA: ${error.message}` };
  }

  revalidatePath('/organizacao');
  return { success: true, message: 'SLA updated successfully!' };
}

/**
 * Delete SLA
 * Story 13.1: Process SLA Management UI
 */
export async function deleteProcessSlaAction(slaId: string): Promise<OrgActionResult> {
  const ctx = await getAuthContext();
  if ('error' in ctx) return { success: false, message: ctx.error };

  const { error } = await ctx.supabase
    .from('org_process_slas')
    .delete()
    .eq('id', slaId)
    .eq('tenant_id', ctx.tenantId);

  if (error) {
    return { success: false, message: `Error deleting SLA: ${error.message}` };
  }

  revalidatePath('/organizacao');
  return { success: true, message: 'SLA deleted successfully!' };
}
```

**Integration (ProcessCockpit360.tsx):**
```typescript
// Add state
const [slaModalOpen, setSlaBmodalOpen] = useState(false);
const [editingSla, setEditingSla] = useState<OrgProcessSla | null>(null);
const [slas, setSlas] = useState<OrgProcessSla[]>([]);

// Fetch SLAs on component load
useEffect(() => {
  const loadSlas = async () => {
    const result = await getProcessSLAsAction(processId);
    if (result.success && result.data) {
      setSlas(result.data);
    }
  };
  loadSlas();
}, [processId]);

// In render, add SLA Management section
<Card>
  <CardHeader>
    <div className="flex justify-between items-center">
      <h3>Definições de SLA</h3>
      <Button onClick={() => setSlaBmodalOpen(true)}>+ Novo SLA</Button>
    </div>
  </CardHeader>
  <CardContent>
    <ProcessSlaList
      processId={processId}
      slas={slas}
      onEdit={(sla) => {
        setEditingSla(sla);
        setSlaBmodalOpen(true);
      }}
      onDelete={async (slaId) => {
        await deleteProcessSlaAction(slaId);
        setSlas(slas.filter(s => s.id !== slaId));
      }}
      onRefresh={() => {
        // Reload SLAs
        const result = await getProcessSLAsAction(processId);
        if (result.success && result.data) {
          setSlas(result.data);
        }
      }}
    />
  </CardContent>
</Card>

// Render modal
<ProcessSlaModal
  isOpen={slaBmodalOpen}
  processId={processId}
  sla={editingSla}
  onClose={() => {
    setSlaBmodalOpen(false);
    setEditingSla(null);
  }}
  onSave={() => {
    // Reload and close
    const result = await getProcessSLAsAction(processId);
    if (result.success && result.data) {
      setSlas(result.data);
    }
    setSlaBmodalOpen(false);
    setEditingSla(null);
  }}
/>
```
