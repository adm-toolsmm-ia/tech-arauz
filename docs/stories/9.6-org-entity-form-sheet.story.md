## Notas Técnicas

### OrgEntityFormSheet Component

```typescript
interface OrgEntityFormSheetProps {
  entity?: 'area' | 'nucleus' | 'process' | 'routine' | 'activity';
  mode: 'create' | 'edit';
  initialData?: any;
  context?: {
    areaId?: string;
    nucleusId?: string;
    processId?: string;
    routineId?: string;
  };
  onSaved?: (saved: any) => void;
  onClose?: () => void;
  isOpen?: boolean;
}

export function OrgEntityFormSheet({ entity, mode, initialData, context, onSaved, ...props }: OrgEntityFormSheetProps) {
  const [isDirty, setIsDirty] = useState(false);
  const [isLoading, setIsLoading] = useState(false);
  const [errors, setErrors] = useState<Record<string, string>>({});
  const [showAlertDialog, setShowAlertDialog] = useState(false);

  // Detecta mudanças
  const handleChange = (field: string, value: any) => {
    setIsDirty(true);
    setFormData({ ...formData, [field]: value });
  };

  // Tenta salvar
  const handleSave = async () => {
    setIsLoading(true);
    try {
      // Chama Server Action baseado em `entity` e `mode`
      const result = await serverActionForEntity(entity, mode, formData, context);
      if (result.success) {
        onSaved?.(result.saved);
        props.onClose?.();
      } else {
        setErrors(result.field_errors || {});
        toast.error(result.error || 'Erro ao salvar');
      }
    } catch (error) {
      toast.error('Erro ao salvar');
    } finally {
      setIsLoading(false);
    }
  };

  // Tenta fechar com dirty check
  const handleClose = () => {
    if (isDirty) {
      setShowAlertDialog(true);
    } else {
      props.onClose?.();
    }
  };

  return (
    <Sheet {...props} onOpenChange={(open) => !open && handleClose()}>
      <SheetContent className="w-full sm:max-w-2xl">
        <SheetHeader>
          <SheetTitle>
            {mode === 'create' ? 'Nova' : 'Editar'} {entityLabel(entity)}
          </SheetTitle>
        </SheetHeader>

        <Tabs defaultValue="info" className="mt-4">
          {/* Tabs por entidade */}
        </Tabs>

        <SheetFooter className="sticky bottom-0 mt-4 gap-2">
          <Button variant="outline" onClick={handleClose}>Cancelar</Button>
          <Button onClick={handleSave} disabled={!isDirty || isLoading}>
            {isLoading ? <Spinner /> : null} Salvar
          </Button>
        </SheetFooter>
      </SheetContent>

      <AlertDialog open={showAlertDialog} onOpenChange={setShowAlertDialog}>
        <AlertDialogContent>
          <AlertDialogTitle>Mudanças não salvas</AlertDialogTitle>
          <AlertDialogDescription>
            Tem mudanças não salvas. Descartar?
          </AlertDialogDescription>
          <AlertDialogCancel>Continuar</AlertDialogCancel>
          <AlertDialogAction onClick={() => {
            setShowAlertDialog(false);
            props.onClose?.();
          }}>
            Descartar
          </AlertDialogAction>
        </AlertDialogContent>
      </AlertDialog>
    </Sheet>
  );
}
```

## Observações

`OrgEntityFormSheet` é o **unificador de CRUD** para toda a hierarquia. Todos os cockpits usam este mesmo componente, apenas com `entity` e `context` diferentes.

Padrão de Server Action dispatch baseado em entity:
```typescript
const serverActions: Record<string, Record<string, Function>> = {
  area: { create: createArea, edit: updateArea },
  nucleus: { create: createNucleus, edit: updateNucleus },
  process: { create: createProcess, edit: updateProcess },
  routine: { create: createRoutine, edit: updateRoutine },
  activity: { create: createActivity, edit: updateActivity },
};

const action = serverActions[entity][mode];
const result = await action(formData, context);
```

Dirty state detection via JSON.stringify é simples, mas pode ser lento em datasets grandes. Se necessário, usar granular tracking por field.
