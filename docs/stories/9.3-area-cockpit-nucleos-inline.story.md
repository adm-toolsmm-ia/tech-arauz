## Notas Técnicas

### Padrão de Tab "Núcleos" em AreaCockpit360

```typescript
<TabsContent value="nucleos">
  <div className="space-y-4">
    <Button onClick={() => openFormSheet('nucleos')}>
      <Plus className="w-4 h-4 mr-2" /> Novo Núcleo
    </Button>
    <div className="grid gap-2">
      {nuclei?.map(nucleus => (
        <OrgEntityCard
          key={nucleus.id}
          title={nucleus.name}
          subtitle={nucleus.objective}
          badge={`${nucleus.processes_count} processos`}
          meta={{
            roles: nucleus.responsible_roles?.length || 0,
            teams: nucleus.teams_count || 0
          }}
          onClick={() => onSelectNucleus?.(nucleus)}
        />
      ))}
    </div>
  </div>
</TabsContent>
```

### Fetch de Processos em NucleusCockpit360

```typescript
const [processes, setProcesses] = useState<OrgProcess[]>([]);
const [loading, setLoading] = useState(false);

useEffect(() => {
  if (!nucleus?.id) return;

  setLoading(true);
  (async () => {
    const result = await getProcessesByNucleus(nucleus.id);
    setProcesses(result || []);
    setLoading(false);
  })();
}, [nucleus?.id]);

if (loading) return <Skeleton className="h-20" />;
```

### State Management em areas-content.tsx

```typescript
const [selectedArea, setSelectedArea] = useState<OrgArea | null>(null);
const [selectedNucleus, setSelectedNucleus] = useState<OrgNucleus | null>(null);

// Invariante: fechar pai fecha filhos
const handleSelectArea = (area: OrgArea) => {
  setSelectedArea(area);
  setSelectedNucleus(null); // Reset filho
};

return (
  <>
    <SplitView
      isOpen={selectedArea !== null}
      onClose={() => handleSelectArea(null)}
    >
      <AreaCockpit360
        area={selectedArea}
        onSelectNucleus={(nucleus) => setSelectedNucleus(nucleus)}
      />
    </SplitView>

    <ContextPanel
      isOpen={selectedNucleus !== null}
      onClose={() => setSelectedNucleus(null)}
      depth={1}
    >
      <NucleusCockpit360
        nucleus={selectedNucleus}
        onSelectProcess={...} // Story 9.4
      />
    </ContextPanel>
  </>
);
```
