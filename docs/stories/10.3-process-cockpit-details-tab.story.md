### ProcessCockpit360 Tab Structure

```typescript
// src/components/organization/ProcessCockpit360.tsx

import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
import { InputsList } from './InputsList';
import { OutputsList } from './OutputsList';
import { RisksList } from './RisksList';
import { ImpactsList } from './ImpactsList';

export interface ProcessCockpit360Props {
  process: OrgProcess;
  onRefresh?: () => Promise<void>;
}

export function ProcessCockpit360({ process, onRefresh }: ProcessCockpit360Props) {
  const [editOpen, setEditOpen] = useState(false);

  const handleSaveEdit = async (data: Partial<OrgProcess>) => {
    await updateProcessAction(process.id, data);
    setEditOpen(false);
    await onRefresh?.();
  };

  return (
    <Tabs defaultValue="overview" className="w-full">
      <TabsList>
        <TabsTrigger value="overview">Visão Geral</TabsTrigger>
        <TabsTrigger value="details">Detalhes</TabsTrigger>
      </TabsList>

      <TabsContent value="overview" className="space-y-4">
        {/* Conteúdo existente de visão geral */}
      </TabsContent>

      <TabsContent value="details" className="space-y-4">
        <div className="flex justify-between items-center">
          <h3 className="text-lg font-semibold">Informações Detalhadas</h3>
          <Button onClick={() => setEditOpen(true)} size="sm">
            Editar
          </Button>
        </div>

        <InputsList inputs={process.inputs} />
        <OutputsList outputs={process.outputs} />
        <RisksList risks={process.risks} />
        <ImpactsList impacts={process.impacts} />
      </TabsContent>

      {editOpen && (
        <OrgEntityFormSheet
          entity={process}
          entityType="process"
          onSave={handleSaveEdit}
          onOpenChange={setEditOpen}
        />
      )}
    </Tabs>
  );
}
```

### InputsList Component

```typescript
// src/components/organization/InputsList.tsx

import { Badge } from '@/components/ui/badge';
import { ArrowRight } from 'lucide-react';

interface InputsListProps {
  inputs: Array<{ name: string; description?: string; required: boolean }>;
}

export function InputsList({ inputs }: InputsListProps) {
  if (!inputs || inputs.length === 0) {
    return null;
  }

  return (
    <div className="space-y-2">
      <h4 className="font-semibold text-sm flex items-center gap-2">
        <ArrowRight className="w-4 h-4" />
        Inputs
      </h4>
      <ul className="space-y-2 ml-6">
        {inputs.map((input) => (
          <li key={input.name} className="text-sm">
            <div className="flex items-center gap-2">
              <span className="font-medium">{input.name}</span>
              {input.required && (
                <Badge variant="outline" className="text-xs">
                  Required
                </Badge>
              )}
            </div>
            {input.description && (
              <p className="text-xs text-muted-foreground mt-1">
                {input.description}
              </p>
            )}
          </li>
        ))}
      </ul>
    </div>
  );
}
```

### RisksList Component (com Severity Badges)

```typescript
// src/components/organization/RisksList.tsx

import { Badge } from '@/components/ui/badge';
import { AlertTriangle } from 'lucide-react';

interface RisksListProps {
  risks: Array<{ name: string; severity?: 'low' | 'medium' | 'high' }>;
}

const severityColors = {
  low: 'bg-blue-100 text-blue-800',
  medium: 'bg-yellow-100 text-yellow-800',
  high: 'bg-red-100 text-red-800',
};

export function RisksList({ risks }: RisksListProps) {
  if (!risks || risks.length === 0) {
    return null;
  }

  return (
    <div className="space-y-2">
      <h4 className="font-semibold text-sm flex items-center gap-2">
        <AlertTriangle className="w-4 h-4 text-destructive" />
        Risks
      </h4>
      <ul className="space-y-2 ml-6">
        {risks.map((risk) => (
          <li key={risk.name} className="text-sm">
            <div className="flex items-center gap-2">
              <span className="font-medium">{risk.name}</span>
              {risk.severity && (
                <Badge className={`text-xs ${severityColors[risk.severity]}`}>
                  {risk.severity}
                </Badge>
              )}
            </div>
          </li>
        ))}
      </ul>
    </div>
  );
}
```
