## 🏗️ Arquitetura

### Schema Structure (src/integrations/espaider/schemas.ts)

```typescript
import { z } from 'zod';

// ============================================================================
// Request Schemas
// ============================================================================

export const ExportarDadosRequestSchema = z.object({
  Token: z.string().min(1, 'Token é obrigatório'),
  Identificador: z.string().min(1, 'Identificador é obrigatório'),
  NumPagina: z.number().int().positive().optional().default(1),
  QuantRegistrosPorPagina: z.number().int().positive().optional().default(100),
  Key: z.string().optional(),
  // BlocoFiltros (dinâmico)
  [key: string]: unknown,
}).strict();

export type ExportarDadosRequest = z.infer<typeof ExportarDadosRequestSchema>;

// ============================================================================
// Response Schemas
// ============================================================================

export const CampoEspaiderSchema = z.object({
  Identificador: z.string(),
  Valor: z.string().nullable(),
}).strict();

export const RegistroEspaiderSchema = z.object({
  IDEspaider: z.number().int(),
  Identificador: z.string(),
  ListaCampos: z.array(CampoEspaiderSchema),
}).strict();

export const URLFilhoSchema = z.object({
  URL: z.string().url('URL inválida'),
  Descricao: z.string().optional(),
  Identificador: z.string().optional(),
}).strict();

export const ExportarDadosResponseSchema = z.object({
  Situacao: z.enum(['S', 'E']).optional(),
  ListaRegistros: z.array(RegistroEspaiderSchema).optional().default([]),
  URLPaginacao: z.string().optional(),
  ListaURLFilhos: z.array(URLFilhoSchema).optional(),
  MensagemRetorno: z.string().optional(),
}).strict();

export type ExportarDadosResponse = z.infer<typeof ExportarDadosResponseSchema>;

// ============================================================================
// Error Response Schema
// ============================================================================

export const EspaiderErrorResponseSchema = z.object({
  Situacao: z.literal('E'),
  MensagemRetorno: z.string(),
}).strict();

export type EspaiderErrorResponse = z.infer<typeof EspaiderErrorResponseSchema>;
```

      throw createError(
        'INVALID_RESPONSE',
        `Espaider error: ${validatedResponse.MensagemRetorno || 'Unknown error'}`,
        requestId,
      );
    }

    return validatedResponse;
  } catch (error) {
    if (error instanceof z.ZodError) {
      // Validation error
      throw createError(
        'INVALID_RESPONSE',
        `API contract violation: ${error.errors.map(e => `${e.path.join('.')}: ${e.message}`).join('; ')}`,
        requestId,
      );

### Contract Tests (src/integrations/espaider/__tests__/contract.test.ts)

import { describe, it, expect } from '@jest/globals';
import { ExportarDadosResponseSchema, RegistroEspaiderSchema } from '../schemas';

describe('Espaider API Contract Tests', () => {
  it('validates real API response (Projects)', () => {
    const realResponse = {
      Situacao: 'S',
      ListaRegistros: [
        {
          IDEspaider: 12345,
          Identificador: 'BI_SOLICITACOES_PROJETOSESPAIDER',
          ListaCampos: [
            { Identificador: 'CODIGO', Valor: 'PROJ-001' },
            { Identificador: 'TITULO', Valor: 'Meu Projeto' },
          ],
        },
      ],
      ListaURLFilhos: [
        {
          URL: 'https://api.espaider.com/...',
          Descricao: 'Entregas',
          Identificador: 'BI_SOLICITACOES_PROJETOSESPAIDER_ENTREGAS',
        },
      ],
    };

    expect(() => ExportarDadosResponseSchema.parse(realResponse)).not.toThrow();
  });

  it('rejects response with extra fields (strict mode)', () => {
    const invalidResponse = {
      Situacao: 'S',
      ListaRegistros: [],
      ExtraField: 'should fail', // Not in schema
    };

    expect(() => ExportarDadosResponseSchema.parse(invalidResponse)).toThrow();
  });

  it('rejects null IDEspaider (must be number)', () => {
    const invalidRecord = {
      IDEspaider: null,
      Identificador: 'test',
      ListaCampos: [],
    };

    expect(() => RegistroEspaiderSchema.parse(invalidRecord)).toThrow();
  });
});

## 📝 Notas Técnicas

### Por que Zod?
- ✅ Tipos derivados (z.infer) eliminam duplicação
- ✅ Runtime validation (detecta mudanças na API)
- ✅ Mensagens de erro claras
- ✅ Já usado em projeto (org_activities validation)

### Schema `.strict()`
- Rejeita campos extras (protege contra API mutations)
- Força que schema evolua se API mudar
- Detecta regressions imediatamente

### Contract Tests
- Fixam respostas reais da API como "contrato"
- Se API mudar, testes falham → alertam
- São fonte de verdade para formato esperado
